const IndexPage = () => {
  const { data: todos, isLoading } = useQuery<Todo[]>(
    ['todos'],
    () => arkzenFetch('/api/a-todo-app/todos').then(r => r.json())
  )

  const [content, setContent] = useState('')

  const addTodo = async () => {
    if (!content.trim()) return
    try {
      await arkzenFetch('/api/a-todo-app/todos', {
        method: 'POST',
        body: JSON.stringify({ content }),
      })
      setContent('')
      refetch()
    } catch {}
  }

  const removeTodo = (id: number) => {
    if (!confirm('Delete this todo?')) return
    arkzenFetch(`/api/a-todo-app/todos/${id}`, { method: 'DELETE' }).then(refetch)
  }

  return (
    <div className="min-h-screen bg-neutral-50 flex items-center justify-center p-6">
      <div className="bg-white rounded-2xl border border-neutral-100 shadow-sm p-8 w-full max-w-md space-y-4">

        {/* Header */}
        <div>
          <h1 className="text-xl font-bold text-neutral-900">Todo Test</h1>
          <p className="text-xs text-neutral-500 mt-0.5">
            CRUD tests for Arkzen v2
          </p>
        </div>

        {/* Input */}
        <div className="relative flex items-center gap-3 w-full">
          <input
            className="flex-1 bg-neutral-50 border border-neutral-200 rounded-xl px-4 py-3 text-sm text-neutral-900 focus:outline-none focus:ring-2 focus:ring-neutral-900/10 focus:border-neutral-400 transition-colors"
            placeholder="Add new todo..."
            value={content}
            onChange={e => setContent(e.target.value)}
          />
          <button
            className="w-12 h-12 rounded-xl bg-neutral-900 flex items-center justify-center text-white text-lg hover:bg-neutral-700 transition-colors"
            onClick={addTodo}
            disabled={!content.trim()}
          >
            + Add
          </button>
        </div>

        {/* List */}
        {isLoading ? (
          <p className="text-xs text-neutral-400">Loading...</p>
        ) : todos?.length === 0 ? (
          <p className="text-xs text-neutral-400">No todos yet.</p>
        ) : (
          <div className="space-y-2">
            {todos.map(todo => (
              <div
                key={todo.id}
                className="flex items-center justify-between gap-3 p-3 rounded-xl bg-white border border-neutral-100 transition-colors hover:shadow-sm"
              >
                <span className={`text-sm flex-1 ${todo.done ? 'line-through text-neutral-400' : 'text-neutral-900'}`}>
                  {todo.content}
                </span>
                <button
                  className="w-8 h-8 rounded-full bg-red-500 flex items-center justify-center text-white text-sm hover:bg-red-600 transition-colors"
                  onClick={() => removeTodo(todo.id)}
                >
                  ×
                </button>
              </div>
            ))}
          </div>
        )}

      </div>
    </div>
  )
}

/* @arkzen:web:page:index:end */
